    from datetime import UTC, datetime

    fields = AgentActivityEvent.__dataclass_fields__

Create `src/agentos/redaction.py` (it must not exist yet):

```python
"""Secret redaction for everything Orin streams or sends away (W0-R11, W7).

Local command hooks and the `orin code` transcript get raw values, as in Claude Code. What leaves through a sink that
other software reads (the activity log and its SSE stream, HTTP hooks, the settings API) goes through these helpers.
They over-redact on purpose: a word like ``author=`` may lose its value, a token never survives.
"""

from __future__ import annotations

import re
from collections.abc import Mapping
from typing import Any

REDACTED = "[REDACTED]"
MAX_DEPTH = 32

_SECRET_WORD = r"(?:token|secret|passw(?:or)?d|passphrase|api[_-]?key|access[_-]?key|private[_-]?key|client[_-]?secret|credential|cookie|session[_-]?id|auth)"
# NAME=value, NAME: value, "NAME": "value", where NAME contains a secret word (GITHUB_TOKEN, db_password, X-Api-Key).
_ASSIGNMENT = re.compile(
    rf"(?i)(?P<name>[\"']?[A-Za-z0-9_.-]*{_SECRET_WORD}[A-Za-z0-9_.-]*[\"']?)(?P<sep>\s*[:=]\s*)"
    r"(?P<value>\"[^\"\r\n]*\"|'[^'\r\n]*'|[^\s,;&|)\]}]+)"
)
# --password x, --token=x, -p is not covered (too ambiguous).
_FLAG = re.compile(rf"(?i)(?P<flag>--?[A-Za-z0-9-]*{_SECRET_WORD}[A-Za-z0-9-]*)(?P<sep>[= ]+)(?P<value>\"[^\"]*\"|'[^']*'|\S+)")
_BEARER = re.compile(r"(?i)\b(bearer|basic|token)\s+[A-Za-z0-9._~+/=-]{8,}")
_URL_PASSWORD = re.compile(r"(?i)\b([a-z][a-z0-9+.-]*://[^/\s:@]+):[^@\s/]+@")
_PRIVATE_KEY = re.compile(r"-----BEGIN [^-]*PRIVATE KEY-----.*?(?:-----END [^-]*PRIVATE KEY-----|$)", re.IGNORECASE | re.DOTALL)
_KNOWN_TOKENS = re.compile(
    r"(?<![A-Za-z0-9])(?:"
    r"gh[pousr]_[A-Za-z0-9]{20,}|github_pat_[A-Za-z0-9_]{20,}|"
    r"sk-[A-Za-z0-9_-]{16,}|xox[abprs]-[A-Za-z0-9-]{10,}|"
    r"AKIA[0-9A-Z]{16}|AIza[0-9A-Za-z_-]{35}|glpat-[A-Za-z0-9_-]{20,}|"
    r"eyJ[A-Za-z0-9_-]{5,}\.[A-Za-z0-9_-]{5,}\.[A-Za-z0-9_-]{5,}"
    r")"
)
_SECRET_KEY = re.compile(rf"(?i){_SECRET_WORD}")
# Mappings whose every value is secret whatever its key (environment blocks, HTTP headers).
_SECRET_CONTAINERS = frozenset({"env", "headers", "environment"})


def redact_text(value: str) -> str:
    """``value`` with every secret-looking part replaced by ``[REDACTED]``."""
    if not value:
        return value
    text = _PRIVATE_KEY.sub(REDACTED, value)
    text = _URL_PASSWORD.sub(lambda match: f"{match.group(1)}:{REDACTED}@", text)
    text = _BEARER.sub(lambda match: f"{match.group(1)} {REDACTED}", text)
    text = _ASSIGNMENT.sub(lambda match: f"{match.group('name')}{match.group('sep')}{REDACTED}", text)
    text = _FLAG.sub(lambda match: f"{match.group('flag')}{match.group('sep')}{REDACTED}", text)
    return _KNOWN_TOKENS.sub(REDACTED, text)


def is_secret_key(key: str) -> bool:
    return bool(_SECRET_KEY.search(str(key)))


def redact_value(value: Any, *, _depth: int = 0, _secret: bool = False) -> Any:
    """A copy of a JSON-like value: strings redacted, values under secret keys (and inside ``env``/``headers``)
    replaced entirely. Deeper than ``MAX_DEPTH`` is replaced too (fail closed)."""
    if _depth > MAX_DEPTH:
        return REDACTED
    if isinstance(value, Mapping):
        result: dict[str, Any] = {}
        for key, item in value.items():
            name = str(key)
            hidden = _secret or is_secret_key(name)
            if name.lower() in _SECRET_CONTAINERS and isinstance(item, Mapping):
                result[name] = {str(inner): REDACTED for inner in item}
            elif hidden and not isinstance(item, (Mapping, list, tuple)):
                result[name] = REDACTED if item not in (None, "", True, False) else item
            else:
                result[name] = redact_value(item, _depth=_depth + 1, _secret=hidden)
        return result
    if isinstance(value, (list, tuple)):
        return [redact_value(item, _depth=_depth + 1, _secret=_secret) for item in value]
    if isinstance(value, str):
        return REDACTED if _secret and value else redact_text(value)
    return value


__all__ = ["REDACTED", "is_secret_key", "redact_text", "redact_value"]
```

def _rule_entries(settings, key: str) -> list[dict[str, Any]]:
    """Every element of ``permissions.<key>`` with its layer and file, highest precedence first (W0-R18). A rule a
    higher layer already lists is kept, marked ``duplicate``."""
    seen: set[str] = set()
    entries: list[dict[str, Any]] = []
    for layer in reversed(settings.layers):
        values = (layer.data.get("permissions") or {}).get(key) if isinstance(layer.data.get("permissions"), dict) else None
        if not isinstance(values, list):
            continue
        for value in values:
            if not isinstance(value, str) or not value.strip():
                continue
            text = value.strip()
            entries.append({"rule": text, "layer": layer.name, "file": str(layer.path) if layer.path else None,
                            "duplicate": text in seen})
            seen.add(text)
    return entries


def _file_report(paths, layer: str) -> list[dict[str, Any]]:
    return [{"path": str(path), "exists": path.is_file()} for path in paths.files_for(layer)]


def effective_report(root: Path | None, trusted: bool) -> dict[str, Any]:
    from agentos.settings.paths import SettingsPaths

    settings = _settings(root, trusted)
    paths = SettingsPaths(root)
    writable: dict[str, str | None] = {}
    for layer in WRITABLE_LAYERS:
        try:
            writable[layer] = str(paths.writable_file(layer))
        except ValueError:
            writable[layer] = None
    return {
        "project_root": str(root) if root is not None else None,
        "trusted": trusted,
        "values": redact_value(dict(settings.data)),
        "provenance": dict(settings.provenance),
        "layers": [{"name": layer.name, "file": str(layer.path) if layer.path else None, "data": redact_value(dict(layer.data))}
                   for layer in settings.layers],
        "files": {layer: _file_report(paths, layer) for layer in ("managed", *WRITABLE_LAYERS)},
        "writable": writable,
        "permissions": {key: _rule_entries(settings, key) for key in RULE_LISTS},
        "errors": [{"file": error.path, "message": error.message} for error in settings.errors],
        "untrusted_ignored": list(settings.untrusted_ignored),
    }


def change_permission_rule(root: Path | None, change: PermissionRuleChange) -> None:
    from agentos.permissions.rules import parse_rule
    from agentos.settings.paths import SettingsPaths
    from agentos.settings.writer import update_settings_file

    rule = change.rule.strip()
    if change.kind == "additionalDirectories":
        if "\n" in rule or "\x00" in rule:
            raise SettingsRejected("invalid_directory", ["a folder is one line"])
    elif change.op == "add" and parse_rule(rule, kind=change.kind, source=change.layer) is None:
        raise SettingsRejected("invalid_rule", [f"'{rule[:200]}' is not a permission rule"])
    try:
        target = SettingsPaths(root).writable_file(change.layer)
    except ValueError as error:
        raise SettingsRejected("project_required", [str(error)]) from error

    def mutate(data: dict[str, Any]) -> None:
        permissions = data.get("permissions")
        if not isinstance(permissions, dict):
            permissions = data["permissions"] = {}
        current = permissions.get(change.kind)
        items = [item for item in current if isinstance(item, str)] if isinstance(current, list) else []
        items = [item for item in items if item.strip() != rule]
        if change.op == "add":
            items.append(rule)
        if items:
            permissions[change.kind] = items
        else:
            permissions.pop(change.kind, None)
        if not permissions:
            data.pop("permissions", None)

    update_settings_file(target, mutate)


def hooks_report(root: Path | None, trusted: bool) -> dict[str, Any]:
    from agentos.hooks.config import load_hook_specs
    from agentos.settings.paths import SettingsPaths

    settings = _settings(root, trusted)
    specs, errors = load_hook_specs(settings)
    paths = SettingsPaths(root)

    }

def _scalar(value: str) -> str:
    return json.dumps(value, ensure_ascii=False)


def write_agent(root: Path | None, name: str, body: AgentFile) -> Path:
    from agentos.agentic.agent_definitions import RESERVED_NAMES, parse_agent_file

    clean = name.strip().lower()
    if not _AGENT_NAME.fullmatch(clean) or clean in RESERVED_NAMES:
        raise SettingsRejected("invalid_agent_name", ["a name of lowercase letters, digits, '-' and '_' that is not a builtin agent"])
    problems = [f"'{tool[:80]}' is not a tool name" for tool in [*(body.tools or []), *body.disallowed_tools] if not _TOOL_NAME.fullmatch(tool.strip())]
    if problems:
        raise SettingsRejected("invalid_agent", problems)
    lines = ["---", f"name: {clean}", f"description: {_scalar(' '.join(body.description.split()))}"]
    if body.tools is not None:
        lines.append(f"tools: {', '.join(tool.strip() for tool in body.tools) or _scalar('')}")
    if body.disallowed_tools:
        lines.append(f"disallowedTools: {', '.join(tool.strip() for tool in body.disallowed_tools)}")
    for key, value in (("model", body.model), ("permissionMode", body.permission_mode), ("effort", body.effort)):
        if value:
            lines.append(f"{key}: {_scalar(value.strip())}")
    if body.max_turns is not None:
        lines.append(f"maxTurns: {body.max_turns}")
    text = "\n".join([*lines, "---", "", body.prompt.strip(), ""])
    folder = _agent_dir(body.scope, root)
    folder.mkdir(parents=True, exist_ok=True)
    target = folder / f"{clean}.md"
    temporary = folder / f".{clean}.md.tmp"
    temporary.write_text(text, encoding="utf-8")
    try:
        parse_agent_file(temporary, origin=body.scope)  # what Orin will read back, checked before it replaces anything
    except (OSError, ValueError) as error:
        temporary.unlink(missing_ok=True)
        raise SettingsRejected("invalid_agent", [str(error)[:300]]) from error
    temporary.replace(target)
    return target


def delete_agent(root: Path | None, name: str, scope: str) -> bool:

    return True

def _log_tail(root: Path, relative: object) -> str:
    if not isinstance(relative, str) or not relative:
        return ""

    try:
        target.relative_to(root.resolve())  # a manifest edited to point outside the workspace is not read
        with target.open("rb") as handle:
            handle.seek(0, 2)
            size = handle.tell()
            handle.seek(max(0, size - LOG_TAIL_BYTES))
            return redact_text(handle.read().decode("utf-8", errors="replace"))
    except (OSError, ValueError):
        return ""

"""Which process a stored pid names, and stopping it only when it is still that process (W6-R3, W7).

A pid alone is not an identity: the OS reuses it. Records that outlive a process (``orin code --bg`` runs, background
shells) keep the creation time read here, and anything that stops one checks it again first.
"""

from __future__ import annotations

import os
import subprocess
from pathlib import Path

_PROCESS_QUERY_LIMITED_INFORMATION = 0x1000
_PROCESS_TERMINATE = 0x0001
_SYNCHRONIZE = 0x00100000


def process_created(pid: int) -> int | str | None:
    """When the process with this pid started, read the same way for a record and for a later check.

    Windows: the creation time from ``GetProcessTimes``; POSIX: field 22 of ``/proc/<pid>/stat``, or where there is
    no ``/proc``, ``ps -o lstart=``. None when it cannot be read.
    """
    if pid <= 0:
        return None
    if os.name == "nt":
        from agentos.hooks.processes import _windows_creation_time, _windows_kernel32

        kernel32 = _windows_kernel32()
        handle = kernel32.OpenProcess(_PROCESS_QUERY_LIMITED_INFORMATION, False, pid)
        if not handle:
            return None
        try:
            return _windows_creation_time(kernel32, handle)
        finally:
            kernel32.CloseHandle(handle)
    stat = Path(f"/proc/{pid}/stat")
    if stat.is_file():
        try:
            fields = stat.read_text(encoding="utf-8").rsplit(")", 1)[1].split()
            return int(fields[19])  # starttime: field 22 counting pid and comm
        except (OSError, IndexError, ValueError):
            return None
    if Path("/proc").is_dir():
        return None  # a system with /proc and no entry for this pid: the process does not exist
    try:
        result = subprocess.run(["ps", "-o", "lstart=", "-p", str(pid)], capture_output=True, text=True, check=False, timeout=10)  # noqa: S603, S607
    except Exception:  # noqa: BLE001 - an identity that cannot be read is None (nothing is ever stopped for it)
        return None
    value = result.stdout.strip()
    return value or None


def same_process(pid: int, created: object) -> bool:
    """True only when ``pid`` is alive and started at ``created``. A missing or unreadable time is never a match."""
    from agentos.launcher.state import process_is_alive

    if pid <= 0 or created is None or created == "" or not process_is_alive(pid):
        return False
    current = process_created(pid)
    return current is not None and current == created


def kill_verified(pid: int, created: object, *, group_only: bool = False) -> bool:
    """Stop the process (and its tree) only if it is still the one that started at ``created``.

    Windows: through a handle opened and validated first, with its validated descendants; never by bare pid.
    POSIX: its process group when it leads one (``group_only`` requires that), else the process itself.
    """
    if not same_process(pid, created):
        return False
    if os.name == "nt":
        return _kill_windows(pid, created)
    import signal

    try:
        if os.getpgid(pid) == pid:
            os.killpg(pid, signal.SIGKILL)
            return True
        if group_only:
            return False
        os.kill(pid, signal.SIGKILL)
    except OSError:
        return False
    return True


def _kill_windows(pid: int, created: object) -> bool:
    from types import SimpleNamespace

    from agentos.hooks.processes import _close_handles, _windows_creation_time, _windows_descendants, _windows_kernel32

    kernel32 = _windows_kernel32()
    handle = kernel32.OpenProcess(_PROCESS_QUERY_LIMITED_INFORMATION | _PROCESS_TERMINATE | _SYNCHRONIZE, False, pid)
    if not handle:
        return False
    descendants: dict[int, int] = {}
    try:
        if _windows_creation_time(kernel32, handle) != created:
            return False  # the pid now belongs to another process
        descendants = _windows_descendants(SimpleNamespace(pid=pid, _handle=handle))
        for child in descendants.values():
            kernel32.TerminateProcess(child, 1)
        return bool(kernel32.TerminateProcess(handle, 1))
    finally:
        _close_handles(descendants)
        kernel32.CloseHandle(handle)

def sweep_expired(root: Path, *, clock=time.time, retention: float = RETENTION_SECONDS) -> None:
    """W6-R9: entries past the retention are removed from every conversation's folder, not only the one being written
    to (a finished conversation's tool results do not stay forever); empty folders go too. At most once an hour."""
    now = clock()
    key = str(root)
    if now - _LAST_SWEEP.get(key, float("-inf")) < SWEEP_EVERY_SECONDS:
        return
    _LAST_SWEEP[key] = now
    cutoff = now - retention
    try:
        folders = [item for item in root.iterdir() if item.is_dir() and not item.is_symlink()]
    except OSError:
        return
    for folder in folders:
        try:
            for item in folder.iterdir():
                try:
                    if item.is_file() and item.stat().st_mtime < cutoff:
                        item.unlink()
                except OSError:
                    continue
            if not any(folder.iterdir()):
                folder.rmdir()
        except OSError:
            continue

/** Small readers for untrusted JSON (W7 API modules): every value is checked, nothing is cast blindly. */

export function rec(value: unknown): Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value) ? (value as Record<string, unknown>) : {}
}

export function str(value: unknown, fallback = ''): string {
  return typeof value === 'string' ? value : fallback
}

export function optStr(value: unknown): string | null {
  return typeof value === 'string' && value.length > 0 ? value : null
}

export function num(value: unknown, fallback = 0): number {
  return typeof value === 'number' && Number.isFinite(value) ? value : fallback
}

export function bool(value: unknown): boolean {
  return value === true
}

export function list<T>(value: unknown, read: (item: unknown) => T | null, max = 1000): T[] {
  if (!Array.isArray(value)) return []
  const items: T[] = []
  for (const item of value.slice(0, max)) {
    const parsed = read(item)
    if (parsed !== null) items.push(parsed)
  }
  return items
}

export function strings(value: unknown, max = 1000): string[] {
  return list(value, (item) => (typeof item === 'string' ? item : null), max)
}

import type { ApiClient } from './client'

export type WritableLayer = 'user' | 'project' | 'local'
export type RuleKind = 'allow' | 'ask' | 'deny' | 'additionalDirectories'

export type RuleEntry = { rule: string; layer: SettingsLayerName; file: string | null; duplicate: boolean }
export type SettingsFile = { path: string; exists: boolean }

export type EffectiveSettings = {
  projectRoot: string | null
  trusted: boolean
  values: Record<string, unknown>
  provenance: Record<string, string>
  layers: Array<{ name: SettingsLayerName; file: string | null; data: Record<string, unknown> }>
  files: Record<string, SettingsFile[]>
  writable: Record<WritableLayer, string | null>
  permissions: Record<RuleKind, RuleEntry[]>
  errors: Array<{ file: string; message: string }>
  untrustedIgnored: string[]
}

const LAYERS: readonly SettingsLayerName[] = ['user', 'project', 'local', 'cli', 'managed']
const KINDS: readonly RuleKind[] = ['allow', 'ask', 'deny', 'additionalDirectories']

function layerName(value: unknown): SettingsLayerName | null {
  return LAYERS.includes(value as SettingsLayerName) ? (value as SettingsLayerName) : null
}

function ruleEntry(value: unknown): RuleEntry | null {
  const data = rec(value)
  const layer = layerName(data.layer)
  const rule = str(data.rule)
  return layer && rule ? { rule, layer, file: optStr(data.file), duplicate: bool(data.duplicate) } : null
}

function permissionsOf(value: unknown): Record<RuleKind, RuleEntry[]> {
  const data = rec(value)
  return Object.fromEntries(KINDS.map((kind) => [kind, list(data[kind], ruleEntry)])) as Record<RuleKind, RuleEntry[]>
}

function stringMap(value: unknown): Record<string, string> {
  return Object.fromEntries(Object.entries(rec(value)).filter((entry): entry is [string, string] => typeof entry[1] === 'string'))
}

export function parseEffectiveSettings(value: unknown): EffectiveSettings {
  const data = rec(value)
  const writable = rec(data.writable)
  return {
    projectRoot: optStr(data.project_root),
    trusted: bool(data.trusted),
    values: rec(data.values),
    provenance: stringMap(data.provenance),
    layers: list(data.layers, (item) => {
      const layer = rec(item)
      const name = layerName(layer.name)
      return name ? { name, file: optStr(layer.file), data: rec(layer.data) } : null
    }),
    files: Object.fromEntries(Object.entries(rec(data.files)).map(([layer, items]) => [
      layer, list(items, (item) => { const file = rec(item); return str(file.path) ? { path: str(file.path), exists: bool(file.exists) } : null }),
    ])),
    writable: { user: optStr(writable.user), project: optStr(writable.project), local: optStr(writable.local) },
    permissions: permissionsOf(data.permissions),
    errors: list(data.errors, (item) => { const error = rec(item); return { file: str(error.file), message: str(error.message) } }),
    untrustedIgnored: strings(data.untrusted_ignored),
  }
}

/**
 * Reading `keybindings.json` and resolving what each key does (W7).
 *
 * The file is a list of `{ context, bindings: { key: action | null } }` blocks, or an object with that list under
 * `bindings` (and optional `$schema`/`$docs`). `null` unbinds a default. Later sources win: Orin's defaults, then
 * Claude Code's file (`~/.claude/keybindings.json`), then Orin's (`~/.orin/keybindings.json`).
 */
import { ACTIONS, BROWSER_RESERVED_KEYS, CLAUDE_CONTEXTS, type Environment, type KeyContext, RESERVED_KEYS, UNSUPPORTED_ACTIONS } from './catalog'
import { parseKeys } from './keys'

export type UserBinding = { context: KeyContext; keys: string[]; action: string | null }
export type Validation = { bindings: UserBinding[]; errors: string[]; warnings: string[] }

const MAX_BINDINGS = 500

export function validateKeybindings(text: string | null, environment: Environment): Validation {
  const result: Validation = { bindings: [], errors: [], warnings: [] }
  if (text === null || !text.trim()) return result
  let data: unknown
  try {
    data = JSON.parse(text)
  } catch (error) {
    result.errors.push(`não é JSON válido: ${error instanceof Error ? error.message : String(error)}`.slice(0, 300))
    return result
  }
  const blocks = Array.isArray(data) ? data : (typeof data === 'object' && data !== null ? (data as Record<string, unknown>).bindings : undefined)
  if (!Array.isArray(blocks)) {
    result.errors.push("o arquivo precisa ser uma lista de blocos, ou um objeto com a lista em 'bindings'")
    return result
  }
  const seen = new Map<string, string>()
  blocks.forEach((block, index) => {
    const entry = typeof block === 'object' && block !== null ? (block as Record<string, unknown>) : null
    const where = `bloco ${index + 1}`
    if (!entry) { result.errors.push(`${where}: não é um objeto`); return }
    const context = entry.context
    if (typeof context !== 'string' || !(CLAUDE_CONTEXTS as readonly string[]).includes(context)) {
      result.errors.push(`${where}: contexto desconhecido '${String(context).slice(0, 60)}'`)
      return
    }
    const map = entry.bindings
    if (typeof map !== 'object' || map === null || Array.isArray(map)) {
      result.errors.push(`${where} (${context}): 'bindings' precisa ser um objeto de tecla → ação`)
      return
    }
    for (const [rawKey, action] of Object.entries(map as Record<string, unknown>)) {
      if (result.bindings.length >= MAX_BINDINGS) { result.errors.push(`mais de ${MAX_BINDINGS} atalhos; o resto foi ignorado`); return }
      const parsed = parseKeys(rawKey)
      if (!parsed.ok) { result.errors.push(`${context}: ${parsed.error}`); continue }
      const label = parsed.keys.join(' ')
      if (action !== null && typeof action !== 'string') { result.errors.push(`${context} ${label}: a ação precisa ser um texto ou null`); continue }
      if (parsed.keys.some((key) => RESERVED_KEYS.has(key))) { result.errors.push(`${context} ${label}: tecla reservada (copiar, colar e edição de texto)`); continue }
      if (typeof action === 'string' && !ACTIONS[action]) {
        if (UNSUPPORTED_ACTIONS.has(action)) result.warnings.push(`${context} ${label}: '${action}' não existe na interface do Orin; o atalho é ignorado`)
        else result.errors.push(`${context} ${label}: ação desconhecida '${action.slice(0, 60)}'`)
        continue
      }
      if (typeof action === 'string' && ACTIONS[action].context !== context && context !== 'Global') {
        result.warnings.push(`${context} ${label}: '${action}' pertence ao contexto ${ACTIONS[action].context}`)
      }
      if (environment === 'browser' && parsed.keys.some((key) => BROWSER_RESERVED_KEYS.has(key))) {
        result.warnings.push(`${context} ${label}: o navegador não deixa a página usar essa tecla; funciona só no app de desktop`)
      }
      const id = `${context}\u0000${label}`
      if (seen.has(id)) result.warnings.push(`${context} ${label}: atalho repetido; vale o último (${String(action)})`)
      seen.set(id, String(action))
      result.bindings.push({ context: context as KeyContext, keys: parsed.keys, action: action as string | null })
    }
  })
  return result
}

/** context → key sequence (canonical, space-joined) → action. */
export type Keymap = Map<KeyContext, Map<string, string>>

export function defaultKeymap(environment: Environment): Keymap {
  const map: Keymap = new Map()
  for (const [action, spec] of Object.entries(ACTIONS)) {
    const keys = environment === 'electron' ? spec.electron : spec.browser
    for (const key of keys) {
      const parsed = parseKeys(key)
      if (!parsed.ok) continue
      if (!map.has(spec.context)) map.set(spec.context, new Map())
      map.get(spec.context)!.set(parsed.keys.join(' '), action)
    }
  }
  return map
}

/** Defaults, then each file's bindings in order (later wins); `null` removes a key. */
export function resolveKeymap(environment: Environment, files: Array<UserBinding[]>): Keymap {
  const map = defaultKeymap(environment)
  for (const bindings of files) {
    for (const binding of bindings) {
      if (!map.has(binding.context)) map.set(binding.context, new Map())
      const sequence = binding.keys.join(' ')
      if (binding.action === null) map.get(binding.context)!.delete(sequence)
      else map.get(binding.context)!.set(sequence, binding.action)
    }
  }
  return map
}

/** The keys bound to an action in the resolved keymap (for hints and the help screen). */
export function keysFor(map: Keymap, action: string): string[][] {
  const context = ACTIONS[action]?.context
  const found: string[][] = []
  for (const [ctx, bindings] of map) {
    if (context && ctx !== context && ctx !== 'Global') continue
    for (const [sequence, bound] of bindings) if (bound === action) found.push(sequence.split(' '))
  }
  return found
}

/**
 * The keybinding engine (W7): which action a key press means in the contexts on screen, chords included, and a hook
 * for surfaces to handle their actions. Text fields keep typing: a handler decides what it takes.
 */
import { createContext, useCallback, useContext, useEffect, useMemo, useRef, useState, type ReactNode } from 'react'
import type { ApiClient } from '../../api/client'
import { getKeybindingsFiles } from '../../api/settings'
import { defaultKeymap, type Keymap, resolveKeymap, validateKeybindings, type Validation } from './bindings'
import type { Environment, KeyContext } from './catalog'
import { CHORD_TIMEOUT_MS, strokeFromEvent } from './keys'

export const PENDING = '__chord_pending__'

export type KeyMatch = string | typeof PENDING | null

/** A stateful matcher over a keymap: feed it key events, it answers an action, PENDING (inside a chord) or null. */
export function createMatcher(keymap: Keymap, clock: () => number = () => Date.now()) {
  let pending: { sequence: string; at: number } | null = null
  function lookup(sequence: string, contexts: readonly KeyContext[]): KeyMatch {
    let prefix = false
    for (const context of contexts) {
      const bindings = keymap.get(context)
      if (!bindings) continue
      const action = bindings.get(sequence)
      if (action) return action
      for (const key of bindings.keys()) if (key.startsWith(`${sequence} `)) prefix = true
    }
    return prefix ? PENDING : null
  }
  return function match(event: Parameters<typeof strokeFromEvent>[0], contexts: readonly KeyContext[]): KeyMatch {
    const stroke = strokeFromEvent(event)
    if (!stroke) return null
    const now = clock()
    if (pending && now - pending.at <= CHORD_TIMEOUT_MS) {
      const continued = lookup(`${pending.sequence} ${stroke}`, contexts)
      if (continued === PENDING) { pending = { sequence: `${pending.sequence} ${stroke}`, at: now }; return PENDING }
      pending = null
      if (continued) return continued
    }
    pending = null
    const single = lookup(stroke, contexts)
    if (single === PENDING) pending = { sequence: stroke, at: now }
    return single
  }
}

export function detectEnvironment(): Environment {
  return typeof window !== 'undefined' && 'orinDesktop' in window ? 'electron' : 'browser'
}

type Handler = (event: KeyboardEvent) => boolean | void
type Registration = { id: number; context: KeyContext; handlers: Record<string, Handler> }

type KeybindingsValue = {
  environment: Environment
  keymap: Keymap
  report: { orin: Validation; claude: Validation }
  match: (event: KeyboardEvent, contexts: readonly KeyContext[]) => KeyMatch
  register: (registration: Omit<Registration, 'id'>) => () => void
  reload: () => void
}

const empty: Validation = { bindings: [], errors: [], warnings: [] }

/** Registrations plus the window listener that sends each key to the newest surface handling its action. */
function createDispatcher(initial: ReturnType<typeof createMatcher>) {
  let matcher = initial
  let registrations: Registration[] = []
  let nextId = 1
  function onKeyDown(event: KeyboardEvent) {
    if (event.defaultPrevented || event.isComposing) return
    // The newest surface first (a dialog over the chat), Global last.
    const stack = [...registrations].reverse()
    const contexts = [...new Set([...stack.map((item) => item.context).filter((context) => context !== 'Global'), 'Global' as const])]
    const action = matcher(event, contexts)
    if (action === PENDING) { event.preventDefault(); return }
    if (!action) return
    for (const item of stack) {
      const handler = item.handlers[action]
      if (handler && handler(event) !== false) { event.preventDefault(); return }
    }
  }
  return {
    setMatcher(next: ReturnType<typeof createMatcher>) { matcher = next },
    register(registration: Omit<Registration, 'id'>) {
      const entry = { ...registration, id: nextId++ }
      registrations = [...registrations, entry]
      return () => { registrations = registrations.filter((item) => item.id !== entry.id) }
    },
    attach() {
      window.addEventListener('keydown', onKeyDown)
      return () => window.removeEventListener('keydown', onKeyDown)
    },
  }
}

const KeybindingsContext = createContext<KeybindingsValue | null>(null)

export function KeybindingsProvider({ client, environment = detectEnvironment(), children }: { client?: ApiClient; environment?: Environment; children: ReactNode }) {
  const [report, setReport] = useState<{ orin: Validation; claude: Validation }>({ orin: empty, claude: empty })
  const [generation, setGeneration] = useState(0)
  useEffect(() => {
    if (!client) return
    let alive = true
    // The files live on the host: a shared server answers 404 and the defaults stay (never an error for the person).
    getKeybindingsFiles(client).then((files) => {
      if (!alive) return
      setReport({ orin: validateKeybindings(files.orin.text, environment), claude: validateKeybindings(files.claude.text, environment) })
    }).catch(() => undefined)
    return () => { alive = false }
  }, [client, environment, generation])
  const keymap = useMemo(
    () => resolveKeymap(environment, [report.claude.bindings, report.orin.bindings]),
    [environment, report],
  )
  const matcher = useMemo(() => createMatcher(keymap), [keymap])
  const [dispatcher] = useState(() => createDispatcher(matcher))
  useEffect(() => { dispatcher.setMatcher(matcher) }, [dispatcher, matcher])
  useEffect(() => dispatcher.attach(), [dispatcher])
  const register = useCallback((registration: Omit<Registration, 'id'>) => dispatcher.register(registration), [dispatcher])
  const value = useMemo<KeybindingsValue>(() => ({
    environment, keymap, report, register, match: matcher, reload: () => setGeneration((value) => value + 1),
  }), [environment, keymap, report, register, matcher])
  return <KeybindingsContext.Provider value={value}>{children}</KeybindingsContext.Provider>
}

// Without a provider (a surface rendered on its own, as in a test), the browser defaults still work: one shared
// dispatcher attaches its listener the first time something registers.
const fallback: KeybindingsValue = (() => {
  const keymap = defaultKeymap('browser')
  const match = createMatcher(keymap)
  const dispatcher = createDispatcher(match)
  let attached = false
  return {
    environment: 'browser', keymap, report: { orin: empty, claude: empty }, match, reload: () => undefined,
    register: (registration) => {
      if (!attached && typeof window !== 'undefined') { dispatcher.attach(); attached = true }
      return dispatcher.register(registration)
    },
  }
})()

export function useKeybindings(): KeybindingsValue {
  return useContext(KeybindingsContext) ?? fallback
}

/** Handle some actions of one context while `enabled`; a handler returning false lets the key through. */
export function useKeybindingHandlers(context: KeyContext, handlers: Record<string, Handler>, enabled = true) {
  const { register } = useKeybindings()
  const latest = useRef(handlers)
  useEffect(() => { latest.current = handlers })
  const actions = Object.keys(handlers).sort().join('|')
  useEffect(() => {
    if (!enabled) return
    const proxy: Record<string, Handler> = {}
    for (const action of actions.split('|').filter(Boolean)) proxy[action] = (event) => latest.current[action]?.(event)
    return register({ context, handlers: proxy })
  }, [context, actions, enabled, register])
}

/**
 * Key strings in Claude Code's `keybindings.json` syntax (W7): modifiers joined with `+` (`ctrl+shift+k`), chords
 * separated by a space (`ctrl+k ctrl+s`). `alt`, `opt`, `option` and `meta` are the Alt/Option key; `cmd`, `command`,
 * `super` and `win` are the Command/Windows key. Names are case-insensitive; the canonical form orders the modifiers
 * ctrl, alt, shift, cmd.
 */

const MODIFIER_ALIASES: Record<string, 'ctrl' | 'alt' | 'shift' | 'cmd'> = {
  ctrl: 'ctrl', control: 'ctrl',
  alt: 'alt', opt: 'alt', option: 'alt', meta: 'alt',
  shift: 'shift',
  cmd: 'cmd', command: 'cmd', super: 'cmd', win: 'cmd',
}
const MODIFIER_ORDER = ['ctrl', 'alt', 'shift', 'cmd'] as const

const KEY_ALIASES: Record<string, string> = {
  esc: 'escape', escape: 'escape', return: 'enter', enter: 'enter', tab: 'tab', space: 'space', ' ': 'space',
  up: 'up', arrowup: 'up', down: 'down', arrowdown: 'down', left: 'left', arrowleft: 'left', right: 'right', arrowright: 'right',
  pageup: 'pageup', pagedown: 'pagedown', home: 'home', end: 'end', backspace: 'backspace', delete: 'delete', del: 'delete',
  insert: 'insert', plus: '+', minus: '-',
}
const NAMED_KEYS = new Set([
  'escape', 'enter', 'tab', 'space', 'up', 'down', 'left', 'right', 'pageup', 'pagedown', 'home', 'end', 'backspace', 'delete',
  'insert', ...Array.from({ length: 12 }, (_, index) => `f${index + 1}`),
])

export const CHORD_TIMEOUT_MS = 3000

export type KeyParse = { ok: true; keys: string[] } | { ok: false; error: string }

/** One keystroke (`ctrl+shift+k`) in canonical form, or null when it is not a valid keystroke. */
export function normalizeStroke(raw: string): string | null {
  const text = raw.trim().toLowerCase()
  if (!text) return null
  // A trailing "+" is the plus key itself ("ctrl++").
  const parts = text.endsWith('++') ? [...text.slice(0, -2).split('+'), '+'] : text.split('+')
  if (parts.some((part) => part === '')) return null
  const key = parts[parts.length - 1]
  const modifiers = new Set<string>()
  for (const part of parts.slice(0, -1)) {
    const modifier = MODIFIER_ALIASES[part]
    if (!modifier || modifiers.has(modifier)) return null
    modifiers.add(modifier)
  }
  const name = KEY_ALIASES[key] ?? key
  if (!NAMED_KEYS.has(name) && [...name].length !== 1) return null
  return [...MODIFIER_ORDER.filter((modifier) => modifiers.has(modifier)), name].join('+')
}

/** A whole binding: one keystroke or a chord of up to three. */
export function parseKeys(raw: string): KeyParse {
  const strokes = raw.trim().split(/\s+/).filter(Boolean)
  if (strokes.length === 0) return { ok: false, error: 'empty key' }
  if (strokes.length > 3) return { ok: false, error: 'a chord has at most three keystrokes' }
  const keys: string[] = []
  for (const stroke of strokes) {
    const normalized = normalizeStroke(stroke)
    if (!normalized) return { ok: false, error: `'${stroke}' is not a keystroke` }
    keys.push(normalized)
  }
  return { ok: true, keys }
}

/** The canonical keystroke of a keyboard event, or null for a lone modifier press. */
export function strokeFromEvent(event: Pick<KeyboardEvent, 'key' | 'ctrlKey' | 'altKey' | 'shiftKey' | 'metaKey'> & { code?: string }): string | null {
  // Option+letter on a Mac types a symbol (Option+P is "π"): with Alt held, the physical key names the letter.
  const physical = event.altKey ? /^Key([A-Z])$/.exec(event.code ?? '')?.[1] ?? /^Digit([0-9])$/.exec(event.code ?? '')?.[1] : undefined
  const raw = physical ? physical.toLowerCase() : event.key
  if (!raw || ['Control', 'Shift', 'Alt', 'Meta', 'AltGraph', 'CapsLock', 'Dead', 'Unidentified'].includes(raw)) return null
  let name = KEY_ALIASES[raw.toLowerCase()] ?? raw.toLowerCase()
  // Shift turns "/" into "?" and "1" into "!": the shifted character is the key, so "shift" is implied, not added,
  // for printable characters (Claude Code's own syntax binds "?" directly).
  const printable = [...name].length === 1 && name !== ' '
  if (name === ' ') name = 'space'
  const modifiers: string[] = []
  if (event.ctrlKey) modifiers.push('ctrl')
  if (event.altKey) modifiers.push('alt')
  if (event.shiftKey && (!printable || /[a-z]/.test(name))) modifiers.push('shift')
  if (event.metaKey) modifiers.push('cmd')
  if (printable && /[a-z]/.test(name)) name = name.toLowerCase()
  return [...modifiers, name].join('+')
}

export function displayKeys(keys: readonly string[], platform: 'mac' | 'other' = 'other'): string {
  const label = (stroke: string) => stroke.split('+').map((part) => {
    if (part === 'ctrl') return platform === 'mac' ? '⌃' : 'Ctrl'
    if (part === 'alt') return platform === 'mac' ? '⌥' : 'Alt'
    if (part === 'shift') return platform === 'mac' ? '⇧' : 'Shift'
    if (part === 'cmd') return platform === 'mac' ? '⌘' : 'Win'
    if (part === 'escape') return 'Esc'
    if (part === 'enter') return 'Enter'
    if (part === 'space') return 'Espaço'
    if (part.length === 1) return part.toUpperCase()
    return part.charAt(0).toUpperCase() + part.slice(1)
  }).join(platform === 'mac' ? '' : '+')
  return keys.map(label).join(' ')
}

export function useChatCommands(deps: ChatCommandDeps) {
  const handlers = useMemo(() => chatCommandHandlers(deps), [deps])
  const offered = useMemo<BuiltinCommand[]>(() => BUILTIN_COMMANDS.filter((command) => command.name in handlers), [handlers])
  async function run(text: string): Promise<SlashOutcome> {
    const parsed = parseSlash(text)
    if (!parsed) return { kind: 'message' }
    const builtin = findBuiltin(parsed.name)
    if (builtin && handlers[builtin.name]) {
      await handlers[builtin.name](parsed.args)
      return { kind: 'handled' }
    }
    if (!builtin && isExcludedCommand(parsed.name)) return { kind: 'excluded', name: parsed.name }
    // A skill, a plugin's command or an MCP prompt: the server expands it (and a built-in not offered yet is sent too).
    return { kind: 'message' }
  }
  return { offered, run }
}

export async function mockChat(page: Page, extra: ExtraRoute = () => undefined): Promise<SentRequest[]> {
  const sent: SentRequest[] = []
  await page.route('**/chats/**', async (route) => {
    if (route.request().resourceType() !== 'document') { await route.fallback(); return }
    const response = await route.fetch()
    const body = (await response.text()).replace('<head>', '<head><meta name="csrf-token" content="csrf-w7-e2e">')
    await route.fulfill({ response, body })
  })
  await page.route('**/v1/**', async (route) => {
    const request = route.request()
    const path = new URL(request.url()).pathname
    const method = request.method()
    let body: unknown
    try { body = request.postDataJSON() } catch { body = undefined }
    sent.push({ method, path, body })
    const custom = extra({ method, path, body })
    if (custom) { await fulfill(route, custom.body, custom.status ?? 200); return }
    if (path.endsWith('/events')) {
      await route.fulfill({ status: 200, contentType: 'text/event-stream', body: 'event: heartbeat\ndata: {"cursor":"1"}\n\n' })
      return
    }
    if (method === 'GET' && path === `/v1/conversations/${CONVERSATION_ID}`) { await fulfill(route, conversation); return }
    if (method === 'GET' && (path === '/v1/conversations' || path === '/v1/projects' || path === '/v1/projects/sidebar' || path === '/v1/skills' || path === '/v1/plugins/commands')) {
      await fulfill(route, { items: [] }); return
    }

}
