## Intentional deviations from the master spec (decided while planning; the ledger repeats them)

1. **Keys stay snake_case.** The spec makes Claude Code names canonical (D5). The registry does make them canonical for everything the model, permission rules, hooks and settings see, and every alias resolves in every mode; but the stored key (`read_file`) stays the internal id. Renaming the key would need a migration of stored transcripts and activity, every UI renderer and ~160 test files at once, and a single missed comparison in a permission or phase check (`EDIT_FAMILY`, `READ_ONLY_TOOLS`, plan-mode lists) would silently let a tool skip a check. Keeping the key removes that whole class of fail-open bug.
2. **Split by mixins.** Each family module holds a mixin with the moved methods; `AgentToolset` inherits them and keeps the constructor and the invoke/permission pipeline (the "context"). There is no separate context object. Module-level helpers move to `agentic/tools/_shared.py`; `agent_tools.py` re-exports them so existing imports keep working.
3. **Two schemas per Claude Code tool.** The public schema comes from the Pydantic `Input` model (Code mode, or any mode with `experimental.claudeToolNames`). Chat mode keeps Orin's legacy schema (`ToolSpec.legacy_schema`) by default. Orin-only tools have no `Input` model.
4. **Input models are not enforced at execution.** They are the schema and the field-name contract; each tool keeps its own validation, whose error messages the existing tests pin.
5. **Output models** are produced for Read, Edit, Write, NotebookEdit, Glob, Grep, Bash, PowerShell, TaskOutput, TaskStop, WebFetch, WebSearch, Cron*, ScheduleWakeup, Monitor and PushNotification as `payload["tool_output"]`, which is also the PostToolUse `tool_response`. Other tools keep the legacy `{status, summary, content}` response.
(Later tasks add deviations 6+ where they arise; each is restated in the task and in the ledger.)

```python
"""The tool contract: one ``ToolSpec`` per tool, collected in a ``ToolRegistry``.

A spec is static: it names the tool, says how to build its ``ToolDefinition``
for one ``AgentToolset`` and, for the tools that mirror Claude Code, carries the
Pydantic ``Input``/``Output`` models whose field names are that public interface.

``key`` is Orin's stable id (``read_file``): stored transcripts, activity events,
the UI and the permission engine use it. ``name`` is the canonical Claude
Code-style name (``Read``) that the model, permission rules and hooks see. Both,
and every alias, resolve to the same spec.
"""
from __future__ import annotations

import copy
from collections.abc import Callable, Iterable, Mapping
from dataclasses import dataclass
from typing import Any

from pydantic import BaseModel

from agentos.agentic.tools.public_names import permission_category, public_name

MAX_RESULT_CHARS_DEFAULT = 12_000


def _always(_toolset: Any) -> bool:
    return True


def _resolve(value: Any, toolset: Any) -> Any:
    return value(toolset) if callable(value) else value


@dataclass(frozen=True, slots=True)
class ToolSpec:
    key: str
    description: str | Callable[[Any], str]
    legacy_schema: Mapping[str, Any] | Callable[[Any], Mapping[str, Any]]
    handler: str
    kind: str
    read_only: bool = False
    policy_tags: tuple[str, ...] | Callable[[Any], tuple[str, ...]] = ()
    max_result_chars: int = MAX_RESULT_CHARS_DEFAULT
    available: Callable[[Any], bool] = _always
    order: int = 0
    aliases: tuple[str, ...] = ()
    input_model: type[BaseModel] | None = None
    public_description: str = ""
    output_model: type[BaseModel] | None = None
    concurrency_safe: bool | None = None

    @property
    def name(self) -> str:
        return public_name(self.key)

    def names(self) -> tuple[str, ...]:
        """Every spelling that resolves to this tool, most specific first."""
        return tuple(dict.fromkeys((self.key, self.name, *self.aliases)))

    def public_schema(self) -> tuple[str, dict[str, Any]] | None:
        if self.input_model is None:
            return None
        return self.public_description, provider_schema(self.input_model)

    def bind(self, toolset: Any) -> Any:
        # Imported here: agent_tools imports the tool modules, which import this one.
        from agentos.agentic.agent_tools import ToolDefinition

        tags = tuple(_resolve(self.policy_tags, toolset))
        return ToolDefinition(
            self.key,
            str(_resolve(self.description, toolset)),
            _resolve(self.legacy_schema, toolset),
            getattr(toolset, self.handler),
            self.kind,
            read_only=self.read_only,
            policy_tags=tags,
            max_result_size_chars=self.max_result_chars,
            public_name=self.name,
            aliases=self.names()[1:],
            permission_category=permission_category(self.key, kind=self.kind, read_only=self.read_only, policy_tags=tags),
            concurrency_safe=self.read_only if self.concurrency_safe is None else self.concurrency_safe,
            input_model=self.input_model,
            output_model=self.output_model,
        )


class ToolRegistryError(ValueError):
    """Two specs claim the same key, name or alias: refused when the registry is built."""


class ToolRegistry:
    def __init__(self, specs: Iterable[ToolSpec]) -> None:
        self._specs: tuple[ToolSpec, ...] = tuple(sorted(specs, key=lambda item: item.order))
        self._by_name: dict[str, ToolSpec] = {}
        self._by_lower: dict[str, ToolSpec] = {}
        for spec in self._specs:
            for spelling in spec.names():
                owner = self._by_lower.get(spelling.lower())
                if owner is not None and owner is not spec:
                    raise ToolRegistryError(f"'{spelling}' names both {owner.key} and {spec.key}")
                self._by_name[spelling] = spec
                self._by_lower[spelling.lower()] = spec

    def specs(self) -> tuple[ToolSpec, ...]:
        return self._specs

    def get(self, name: str) -> ToolSpec | None:
        if not isinstance(name, str) or not name or name.startswith("mcp__"):
            return None
        return self._by_name.get(name) or self._by_lower.get(name.lower())

    def key_for(self, name: str) -> str | None:
        spec = self.get(name)
        return spec.key if spec is not None else None


# -- provider-safe JSON Schema from a Pydantic model -------------------------

_DROPPED_KEYS = frozenset({"title", "default", "examples"})


def provider_schema(model: type[BaseModel]) -> dict[str, Any]:
    """A JSON Schema every provider accepts: no titles, no $ref, optional fields as plain types."""
    raw = model.model_json_schema(by_alias=True, mode="validation")
    definitions = raw.pop("$defs", {})
    return _clean(raw, definitions)


def _clean(node: Any, definitions: Mapping[str, Any]) -> Any:
    if isinstance(node, list):
        return [_clean(item, definitions) for item in node]
    if not isinstance(node, Mapping):
        return node
    if "$ref" in node:
        target = definitions.get(str(node["$ref"]).rsplit("/", 1)[-1], {})
        merged = {**copy.deepcopy(target), **{k: v for k, v in node.items() if k != "$ref"}}
        return _clean(merged, definitions)
    options = node.get("anyOf")
    if isinstance(options, list):
        concrete = [item for item in options if not (isinstance(item, Mapping) and item.get("type") == "null")]
        if len(concrete) == 1 and len(concrete) < len(options):
            rest = {k: v for k, v in node.items() if k != "anyOf"}
            return _clean({**copy.deepcopy(concrete[0]), **rest}, definitions)
    cleaned: dict[str, Any] = {}
    for key, value in node.items():
        if key in _DROPPED_KEYS:
            continue
        if key == "properties" and isinstance(value, Mapping):
            cleaned[key] = {name: _clean(item, definitions) for name, item in value.items()}
        else:
            cleaned[key] = _clean(value, definitions)
    if cleaned.get("type") == "object" and "properties" in cleaned:
        cleaned.setdefault("required", [])
        cleaned.setdefault("additionalProperties", False)
    return cleaned
```

"""Mechanical split of ``AgentToolset`` (W4 plan, Task 2 and Task 3).

Usage (from the repository root):
    uv run python <this file> shared
    uv run python <this file> move <module> <MixinClass> <method>[,<method>...]

``shared`` moves every top-level statement between the import block and
``class AgentToolset`` from ``agent_tools.py`` into
``agentos/agentic/tools/_shared.py`` and re-imports the names.

``move`` moves the named ``AgentToolset`` methods, verbatim, into a new mixin
class in ``agentos/agentic/tools/<module>.py`` (dotted, e.g. ``files.read``)
and adds the mixin to ``AgentToolset``'s bases.

The script refuses (exit 2, nothing written) when a method is missing, when a
method uses a name the new module cannot import without a cycle (anything
defined after the class, or ``AgentToolset`` itself), or when the target
module already exists.
"""
from __future__ import annotations

import ast
import builtins
import sys
from pathlib import Path

ROOT = Path.cwd()
TOOLS = ROOT / "src" / "agentos" / "agentic" / "agent_tools.py"
PACKAGE = ROOT / "src" / "agentos" / "agentic" / "tools"
SHARED = PACKAGE / "_shared.py"
SHARED_IMPORT_MARK = "from agentos.agentic.tools._shared import ("


def _group(module: str) -> int:
    top = module.split(".")[0]
    if top == "agentos":
        return 2
    return 0 if top in sys.stdlib_module_names else 1


def tidy_imports(text: str) -> str:
    """One ``from X import`` per module, grouped stdlib / third-party / agentos, names in first-seen order.

    Only the contiguous import block after the docstring and ``from __future__`` is rewritten.
    """
    tree = ast.parse(text)
    rows = text.splitlines(keepends=True)
    block: list[ast.stmt] = []
    for node in tree.body:
        if not block and isinstance(node, ast.Expr) and isinstance(getattr(node, "value", None), ast.Constant):
            continue
        if not block and isinstance(node, ast.ImportFrom) and node.module == "__future__":
            continue
        if isinstance(node, (ast.Import, ast.ImportFrom)):
            block.append(node)
            continue
        break
    if not block:
        return text
    first, last = block[0].lineno - 1, block[-1].end_lineno
    froms: dict[str, list[str]] = {}
    plain: list[tuple[str, str]] = []  # (module, rendered `import x` line)
    for node in block:
        if isinstance(node, ast.ImportFrom) and node.level == 0:
            names = froms.setdefault(node.module, [])
            for alias in node.names:
                rendered = alias.name + (f" as {alias.asname}" if alias.asname else "")
                if rendered not in names:
                    names.append(rendered)
        else:
            rendered = "".join(rows[node.lineno - 1: node.end_lineno]).rstrip("\n")
            module = node.names[0].name if isinstance(node, ast.Import) else str(node.module or "")
            if rendered not in [line for _, line in plain]:
                plain.append((module, rendered))
    lines_by_group: dict[int, list[str]] = {0: [], 1: [], 2: []}
    for module, rendered in plain:
        lines_by_group[_group(module)].append(rendered)
    for module, names in froms.items():
        line = f"from {module} import {', '.join(names)}"
        if len(line) > 110:
            line = f"from {module} import (\n" + "".join(f"    {name},\n" for name in names) + ")"
        lines_by_group[_group(module)].append(line)
    groups = ["\n".join(lines) for _, lines in sorted(lines_by_group.items()) if lines]
    return "".join(rows[:first]) + "\n\n".join(groups) + "\n" + "".join(rows[last:])


def fail(message: str) -> None:
    print(f"REFUSED: {message}")
    sys.exit(2)


def absolute(module: str | None, level: int) -> str:
    """``.diagnostics`` seen from agentos.agentic -> ``agentos.agentic.diagnostics``."""
    if level == 0:
        return module or ""
    base = ["agentos", "agentic"][: max(0, 2 - (level - 1))]
    return ".".join(base + ([module] if module else []))


def import_text(node: ast.stmt, keep: set[str] | None = None) -> str | None:
    """Re-render an import statement with absolute modules, optionally keeping only ``keep`` bound names."""
    if isinstance(node, ast.Import):
        names = [alias for alias in node.names if keep is None or (alias.asname or alias.name.split(".")[0]) in keep]
        if not names:
            return None
        return "import " + ", ".join(alias.name + (f" as {alias.asname}" if alias.asname else "") for alias in names)
    assert isinstance(node, ast.ImportFrom)
    if node.module == "__future__":
        return None
    names = [alias for alias in node.names if keep is None or (alias.asname or alias.name) in keep]
    if not names:
        return None
    rendered = ", ".join(alias.name + (f" as {alias.asname}" if alias.asname else "") for alias in names)
    module = absolute(node.module, node.level)
    if len(rendered) > 100:
        inner = "".join(f"    {alias.name}{f' as {alias.asname}' if alias.asname else ''},\n" for alias in names)
        return f"from {module} import (\n{inner})"
    return f"from {module} import {rendered}"


def bound_by_import(node: ast.stmt) -> list[str]:
    if isinstance(node, ast.Import):
        return [alias.asname or alias.name.split(".")[0] for alias in node.names]
    if isinstance(node, ast.ImportFrom):
        return [alias.asname or alias.name for alias in node.names]
    return []


def defined_names(node: ast.stmt) -> list[str]:
    if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
        return [node.name]
    if isinstance(node, ast.Assign):
        out: list[str] = []
        for target in node.targets:
            out += [item.id for item in ast.walk(target) if isinstance(item, ast.Name)]
        return out
    if isinstance(node, (ast.AnnAssign, ast.AugAssign)) and isinstance(node.target, ast.Name):
        return [node.target.id]
    return []


def load_names(nodes: list[ast.AST]) -> set[str]:
    names: set[str] = set()
    for node in nodes:
        for item in ast.walk(node):
            if isinstance(item, ast.Name) and isinstance(item.ctx, ast.Load):

    return names

    # delete bottom-up so earlier line numbers stay valid

NEW_BUILDER = '''    def _build_definitions(self) -> tuple[ToolDefinition, ...]:

def _cut(line: str, start: int | None = None, end: int | None = None) -> str:
    """Slice ``line`` by UTF-8 byte offsets, the unit ``ast`` column offsets use."""
    return line.encode("utf-8")[start:end].decode("utf-8")


def char_col(source_lines: list[str], lineno: int, col: int) -> int:
    return len(_cut(source_lines[lineno - 1], 0, col))


def segment(source_lines: list[str], node: ast.AST) -> str:
    """Exact source of ``node`` (may span lines)."""
    first, last = node.lineno - 1, node.end_lineno - 1
    if first == last:
        return _cut(source_lines[first], node.col_offset, node.end_col_offset)
    parts = [_cut(source_lines[first], node.col_offset)]
    parts += source_lines[first + 1:last]
    parts.append(_cut(source_lines[last], 0, node.end_col_offset))
    return "".join(parts)


def substitute(source_lines: list[str], node: ast.AST, locals_: dict[str, str]) -> tuple[str, bool]:
    """``node``'s source with every local variable replaced by its replacement text; True if any was."""
    spans = [
        item for item in ast.walk(node)
        if isinstance(item, ast.Name) and isinstance(item.ctx, ast.Load) and item.id in locals_
    ]
    text = segment(source_lines, node)
    if not spans:
        return text, False
    lines = [line.encode("utf-8") for line in text.splitlines(keepends=True)]
    for item in sorted(spans, key=lambda n: (n.lineno, n.col_offset), reverse=True):
        row = item.lineno - node.lineno
        start = item.col_offset - (node.col_offset if row == 0 else 0)
        end = item.end_col_offset - (node.col_offset if row == 0 else 0)
        lines[row] = lines[row][:start] + locals_[item.id].encode("utf-8") + lines[row][end:]
    return b"".join(lines).decode("utf-8"), True


def reindent(text: str, original_col: int, new_col: int) -> str:
    lines = text.splitlines(keepends=True)
    if len(lines) <= 1:
        return text
    delta = original_col - new_col
    out = [lines[0]]
    for line in lines[1:]:
        if delta > 0:
            strip = min(delta, len(line) - len(line.lstrip(" ")))
            out.append(line[strip:])
        else:
            out.append(" " * (-delta) + line)
    return "".join(out)


def uses_self(node: ast.AST) -> bool:
    return any(isinstance(item, ast.Name) and item.id == "self" for item in ast.walk(node))


def method_modules() -> dict[str, str]:
    """handler method name -> dotted module under agentos.agentic.tools that defines it."""
    owners: dict[str, str] = {}
    for path in PACKAGE.rglob("*.py"):
        if path.name.startswith("_") or path.name in {"public_names.py", "spec.py", "registry.py"}:
            continue
        tree = ast.parse(path.read_text(encoding="utf-8"))
        dotted = ".".join(path.relative_to(PACKAGE).with_suffix("").parts)
        for node in tree.body:
            if isinstance(node, ast.ClassDef):
                for item in node.body:
                    if isinstance(item, (ast.FunctionDef, ast.AsyncFunctionDef)):
                        owners[item.name] = dotted
    return owners


def specs() -> None:
    registry_path = PACKAGE / "registry.py"
    if registry_path.exists():
        fail("registry.py already exists")
    source = TOOLS.read_text(encoding="utf-8")
    source_lines = source.splitlines(keepends=True)
    tree = ast.parse(source)
    cls = class_node(tree)
    builder = next((n for n in cls.body if isinstance(n, ast.FunctionDef) and n.name == "_build_definitions"), None)
    if builder is None:
        fail("_build_definitions not found")
    if "self._mcp_provider.definitions()" not in segment(source_lines, builder):
        fail("the builder no longer appends the MCP provider's definitions as expected")
    owners = method_modules()
    imports = {}
    for node in tree.body:
        for bound in bound_by_import(node):
            imports[bound] = node
    from_shared = shared_names(tree)
    # local name -> (replacement text, evaluated at bind time?, names the replacement uses)
    hoisted: dict[str, tuple[str, set[str]]] = {}  # constant name -> (definition text, names used)
    entries: list[tuple[str, str, set[str], set[str]]] = []  # (module, ToolSpec text, names used, constants used)
    counter = 0

    def names_in(node: ast.AST) -> set[str]:
        return {n.id for n in ast.walk(node) if isinstance(n, ast.Name) and isinstance(n.ctx, ast.Load)}

    def pure(node: ast.AST, scope: dict[str, tuple]) -> bool:
        """A value that can be computed once at import: no self, no locals, only calls of shared helpers."""
        if uses_self(node) or names_in(node) & set(scope):
            return False
        for item in ast.walk(node):
            if isinstance(item, ast.Call) and not (isinstance(item.func, ast.Name) and item.func.id in from_shared):
                return False
        return True

    def visit(statements: list[ast.stmt], guards: list[str], scope: dict[str, tuple]) -> None:
        nonlocal counter
        scope = dict(scope)
        for statement in statements:
            if isinstance(statement, ast.If):
                test, _ = substitute(source_lines, statement.test, {k: v[0] for k, v in scope.items()})
                visit(statement.body, guards + [test], scope)
                if statement.orelse:
                    visit(statement.orelse, guards + [f"not ({test})"], scope)
                continue
            target = None
            if isinstance(statement, ast.Assign) and len(statement.targets) == 1 and isinstance(statement.targets[0], ast.Name):
                target = statement.targets[0].id
            elif isinstance(statement, ast.AnnAssign) and isinstance(statement.target, ast.Name) and statement.value is not None:
                target = statement.target.id
            if target is not None and target not in {"items", "native"}:
                name = target
                if pure(statement.value, scope):
                    constant = "_" + name.upper()
                    text = reindent(segment(source_lines, statement.value), char_col(source_lines, statement.lineno, statement.col_offset), 0)
                    hoisted[constant] = (text, names_in(statement.value))
                    scope[name] = (constant, False, {constant})
                else:
                    text, _ = substitute(source_lines, statement.value, {k: v[0] for k, v in scope.items()})
                    text = " ".join(part.strip() for part in text.splitlines())
                    used = names_in(statement.value) - set(scope)
                    for other in names_in(statement.value) & set(scope):
                        used |= scope[other][2]
                    scope[name] = (f"({text})", True, used)
                continue
            for call in ast.walk(statement):
                if isinstance(call, ast.Call) and isinstance(call.func, ast.Name) and call.func.id == "ToolDefinition":
                    counter += 10
                    entries.append(convert(call, guards, scope, counter))

    def convert(call: ast.Call, guards: list[str], scope: dict[str, tuple], order_value: int):
        args = list(call.args)
        keywords = {item.arg: item.value for item in call.keywords}
        labels = ["name", "description", "parameters", "handler", "kind"]
        values: dict[str, ast.AST] = dict(zip(labels, args))
        for label in labels:
            if label in keywords:
                values[label] = keywords.pop(label)
        if set(values) != set(labels):
            fail(f"ToolDefinition at line {call.lineno} lacks one of {labels}")
        key, handler, kind = values["name"], values["handler"], values["kind"]
        if not (isinstance(key, ast.Constant) and isinstance(key.value, str)):
            fail(f"ToolDefinition at line {call.lineno} has a non-literal name")
        if not (isinstance(handler, ast.Attribute) and isinstance(handler.value, ast.Name) and handler.value.id == "self"):
            fail(f"{key.value}: handler is not self.<method>")
        if handler.attr not in owners:
            fail(f"{key.value}: handler {handler.attr} was not moved to a tools module")
        if not (isinstance(kind, ast.Constant) and isinstance(kind.value, str)):
            fail(f"{key.value}: kind is not a literal")
        used: set[str] = set()
        constants: set[str] = set()
        fields = [f'key="{key.value}"']

        def field_text(label: str, node: ast.AST) -> str:
            replacements = {name: value[0] for name, value in scope.items()}
            text, _ = substitute(source_lines, node, replacements)
            late = uses_self(node)
            for name in names_in(node):
                if name in scope:
                    late = late or scope[name][1]
                    (used if scope[name][1] else constants).update(scope[name][2])
                else:
                    used.add(name)
            text = reindent(text, char_col(source_lines, node.lineno, node.col_offset), SPEC_INDENT)
            if not late:
                return f"{label}={text}"
            if "\n" not in text:
                return f"{label}=lambda self: {text}"
            inner = reindent(text, SPEC_INDENT, SPEC_INDENT + 4)
            return f"{label}=lambda self: (\n{' ' * (SPEC_INDENT + 4)}{inner}\n{' ' * SPEC_INDENT})"

        fields.append(field_text("description", values["description"]))
        fields.append(field_text("legacy_schema", values["parameters"]))
        fields.append(f'handler="{handler.attr}"')
        fields.append(f'kind="{kind.value}"')
        for keyword, label in (("read_only", "read_only"), ("policy_tags", "policy_tags"), ("max_result_size_chars", "max_result_chars")):
            if keyword in keywords:
                fields.append(field_text(label, keywords.pop(keyword)))
        if keywords:
            fail(f"{key.value}: unexpected keywords {sorted(keywords)}")
        if guards:
            guard_text = " and ".join(f"({item})" for item in guards)
            used |= names_in(ast.parse(guard_text, mode="eval"))
            fields.append(f"available=lambda self: {guard_text}")
        fields.append(f"order={order_value}")
        used -= {"self"}
        used -= set(hoisted)
        body = "    ToolSpec(\n" + "".join(f"{' ' * SPEC_INDENT}{text},\n" for text in fields) + "    ),\n"
        return owners[handler.attr], body, used, constants

    visit(builder.body, [], {})
    if not entries:
        fail("no ToolDefinition found")
    by_module: dict[str, list[tuple[str, set[str], set[str]]]] = {}
    for module, text, used, constants in entries:
        by_module.setdefault(module, []).append((text, used, constants))
    pending: dict[Path, str] = {}
    for module, items in by_module.items():
        path = PACKAGE.joinpath(*module.split(".")).with_suffix(".py")
        text = path.read_text(encoding="utf-8")
        mod_tree = ast.parse(text)
        bound: set[str] = set()
        last_import_end = 0
        for node in mod_tree.body:
            if isinstance(node, (ast.Import, ast.ImportFrom)):
                bound.update(bound_by_import(node))
                last_import_end = node.end_lineno
        module_constants = sorted(set().union(*(constants for _, _, constants in items)))
        needed = set().union(*(used for _, used, _ in items))
        for constant in module_constants:
            needed |= hoisted[constant][1]
        needed -= bound | {"ToolSpec", "self"} | set(dir(builtins)) | set(module_constants)
        shared_needed = sorted(name for name in needed if name in from_shared)
        other = sorted(name for name in needed if name not in from_shared)
        unknown = [name for name in other if name not in imports]
        if unknown:
            fail(f"{module}: names with no known origin in specs: {unknown}")
        lines: list[str] = []
        seen: set[int] = set()
        for node in tree.body:
            if id(node) in seen or not any(imports.get(name) is node for name in other):
                continue
            seen.add(id(node))
            rendered = import_text(node, {name for name in bound_by_import(node) if name in other})
            if rendered:
                lines.append(rendered)
        if shared_needed:
            lines.append("from agentos.agentic.tools._shared import (\n" + "".join(f"    {name},\n" for name in shared_needed) + ")")
        lines.append("from agentos.agentic.tools.spec import ToolSpec")
        text_lines = text.splitlines(keepends=True)
        text_lines.insert(last_import_end, "\n".join(lines) + "\n")
        block = "\n\n# Tool specs, moved from AgentToolset._build_definitions (W4).\n"
        for constant in module_constants:
            block += f"{constant} = {hoisted[constant][0]}\n\n"
        block += "SPECS: tuple[ToolSpec, ...] = (\n" + "".join(item for item, _, _ in items) + ")\n"
        new_text = "".join(text_lines).rstrip() + "\n" + block
        try:
            ast.parse(new_text)
        except SyntaxError as error:
            bad = new_text.splitlines()
            context = "\n".join(bad[max(0, (error.lineno or 1) - 8): (error.lineno or 1) + 2])
            fail(f"{module}: generated code does not parse at line {error.lineno}:\n{context}")
        pending[path] = new_text
        print(f"{module}: {len(items)} specs")
    # Nothing is written unless every module parsed.
    for path, new_text in pending.items():
        path.write_text(tidy_imports(new_text), encoding="utf-8")
    modules = list(by_module)
    alias = {module: "_" + module.replace(".", "_") for module in modules}
    registry_text = '"""Every built-in tool spec, in the order the model has always seen them (W4)."""\n'
    registry_text += "from __future__ import annotations\n\nfrom functools import lru_cache\n\n"
    registry_text += "".join(f"from agentos.agentic.tools.{module} import SPECS as {alias[module]}\n" for module in modules)
    registry_text += "from agentos.agentic.tools.spec import ToolRegistry\n\n\n"
    registry_text += "@lru_cache(maxsize=1)\ndef default_registry() -> ToolRegistry:\n    return ToolRegistry((\n"
    registry_text += "".join(f"        *{alias[module]},\n" for module in modules)
    registry_text += "    ))\n"
    registry_path.write_text(registry_text, encoding="utf-8")
    start = first_line(builder) - 1
    end = builder.end_lineno
    new_source = "".join(source_lines[:start]) + NEW_BUILDER + "".join(source_lines[end:])
    ast.parse(new_source)
    TOOLS.write_text(new_source, encoding="utf-8")
    print(f"{len(entries)} specs in {len(modules)} modules; registry written")


def main() -> None:

def _group(module: str) -> int:
    top = module.split(".")[0]
    if top == "agentos":
        return 2
    return 0 if top in sys.stdlib_module_names else 1


def tidy_imports(text: str) -> str:
    """One ``from X import`` per module, grouped stdlib / third-party / agentos, names in first-seen order.

    Only the contiguous import block after the docstring and ``from __future__`` is rewritten.
    """
    tree = ast.parse(text)
    rows = text.splitlines(keepends=True)
    block: list[ast.stmt] = []
    for node in tree.body:
        if not block and isinstance(node, ast.Expr) and isinstance(getattr(node, "value", None), ast.Constant):
            continue
        if not block and isinstance(node, ast.ImportFrom) and node.module == "__future__":
            continue
        if isinstance(node, (ast.Import, ast.ImportFrom)):
            block.append(node)
            continue
        break
    if not block:
        return text
    first, last = block[0].lineno - 1, block[-1].end_lineno
    froms: dict[str, list[str]] = {}
    plain: list[tuple[str, str]] = []  # (module, rendered `import x` line)
    for node in block:
        if isinstance(node, ast.ImportFrom) and node.level == 0:
            names = froms.setdefault(node.module, [])
            for alias in node.names:
                rendered = alias.name + (f" as {alias.asname}" if alias.asname else "")
                if rendered not in names:
                    names.append(rendered)
        else:
            rendered = "".join(rows[node.lineno - 1: node.end_lineno]).rstrip("\n")
            module = node.names[0].name if isinstance(node, ast.Import) else str(node.module or "")
            if rendered not in [line for _, line in plain]:
                plain.append((module, rendered))
    lines_by_group: dict[int, list[str]] = {0: [], 1: [], 2: []}
    for module, rendered in plain:
        lines_by_group[_group(module)].append(rendered)
    for module, names in froms.items():
        line = f"from {module} import {', '.join(names)}"
        if len(line) > 110:
            line = f"from {module} import (\n" + "".join(f"    {name},\n" for name in names) + ")"
        lines_by_group[_group(module)].append(line)
    groups = ["\n".join(lines) for _, lines in sorted(lines_by_group.items()) if lines]
    return "".join(rows[:first]) + "\n\n".join(groups) + "\n" + "".join(rows[last:])


def fail(message: str) -> None:
    print(f"REFUSED: {message}")
    sys.exit(2)


NEEDED = (
    ("typing", ("Any", "Literal")),
    ("pydantic", ("Field",)),
    ("agentos.agentic.tools.spec", ("ToolInput",)),
)


def add_imports(text: str) -> str:
    """Add the imports the models need after the module's last top-level import, skipping bound names."""
    tree = ast.parse(text)
    bound: set[str] = set()
    last = 0
    for node in tree.body:
        if isinstance(node, ast.ImportFrom):
            bound.update(alias.asname or alias.name for alias in node.names)
            last = node.end_lineno
        elif isinstance(node, ast.Import):
            bound.update((alias.asname or alias.name).split(".")[0] for alias in node.names)
            last = node.end_lineno
    lines = [
        f"from {module} import {', '.join(name for name in names if name not in bound)}\n"
        for module, names in NEEDED
        if any(name not in bound for name in names)
    ]
    rows = text.splitlines(keepends=True)
    return "".join(rows[:last] + lines + rows[last:])


def main() -> None:
    sys.path.insert(0, str(Path.cwd() / "src"))
    from agentos.agentic.tools import public_names

    descriptions = {key: entry[0] for key, entry in getattr(public_names, "_PUBLIC_SCHEMAS", {}).items()}
    missing = sorted(set(ATTACH) - set(descriptions))
    if missing:
        fail(f"no public description for {missing} (run this before _PUBLIC_SCHEMAS is removed)")
    pending: dict[Path, str] = {}
    for module, source in MODELS.items():
        path = PACKAGE.joinpath(*module.split(".")).with_suffix(".py")
        text = pending.get(path) or path.read_text(encoding="utf-8")
        if MARKER not in text:
            fail(f"{module}: spec marker missing")
        text = text.replace(MARKER, source.strip() + "\n\n\n" + MARKER, 1)
        pending[path] = add_imports(text)
    for key, (module, model) in ATTACH.items():
        path = PACKAGE.joinpath(*module.split(".")).with_suffix(".py")
        text = pending[path]
        anchor = f'        key="{key}",\n'
        start = text.find(anchor)
        if start < 0:
            fail(f"{key}: ToolSpec not found in {module}")
        kind_at = text.find("        kind=", start)
        line_end = text.find("\n", kind_at)
        if kind_at < 0 or line_end < 0:
            fail(f"{key}: kind line not found")
        addition = f"        input_model={model},\n        public_description={descriptions[key]!r},\n"
        text = text[: line_end + 1] + addition + text[line_end + 1:]
        pending[path] = text
    for path, text in pending.items():
        try:
            ast.parse(text)
        except SyntaxError as error:
            fail(f"{path.name}: does not parse ({error})")
    for path, text in pending.items():
        path.write_text(tidy_imports(text), encoding="utf-8")
    print(f"attached {len(ATTACH)} models in {len(MODELS)} modules")


if __name__ == "__main__":
    main()
